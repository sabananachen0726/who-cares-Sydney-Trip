    function compressImage(file) {

      return new Promise((resolve, reject) => {

        const reader = new FileReader();

        reader.readAsDataURL(file);

        reader.onload = (e) => {

          const img = new Image();

          img.src = e.target.result;

          img.onload = () => {

            const canvas = document.createElement("canvas");

            let width = img.width, height = img.height;

            if (width > 1024) { height = Math.round((height * 1024) / width); width = 1024; }

            canvas.width = width; canvas.height = height;

            const ctx = canvas.getContext("2d");

            ctx.drawImage(img, 0, 0, width, height);

            resolve(canvas.toDataURL("image/jpeg", 0.8).split(',')[1]);

          };

        };

      });

    }


    async function analyzeReceiptWithGemini(base64Data) {

      const prompt = `

        You are an expert expense parser for a trip to Sydney, Australia.

        Analyze this receipt. EXTRACT ALL FIELDS IN ORIGINAL ENGLISH AS SEEN ON THE RECEIPT (DO NOT TRANSLATE stores or items into Chinese).

        Return purely a strict raw JSON matching this format:

        {

          "storeName": "Original English name of store (e.g. Woolworths Metro, Coles, Cafe Name)",

          "items": "Concise summary line of bought items in English (e.g. Milk, Avocado Toast, Coffee)",

          "amountAUD": Grand total price numeric float (e.g. 24.50, exclude changes/cash given metrics),

          "category": "Pick one exactly: Food, Supermarket, Transport, Tickets, Hotel, Shopping, Others",

          "date": "Date found on receipt in YYYY-MM-DD. If completely missing, use today: ${new Date().toISOString().split('T')[0]}"

        }

      `;

      const result = await modelCallWithFallback(prompt, base64Data);

      return JSON.parse(result);

    }


    async function modelCallWithFallback(prompt, base64Data) {

      // Fallback matrix routing logic

      const models = ["gemini-2.0-flash", "gemini-1.5-flash"];

      for (let modelName of models) {

        try {

          const result = await aiModel.generateContent([prompt, { inlineData: { data: base64Data, mimeType: "image/jpeg" } }]);

          return result.response.text();

        } catch (e) {

          console.warn(`Model ${modelName} encountered error. Trying fallback...`);

        }

      }

      throw new Error("All active Gemini endpoints failed.");

    }

      paidBySel.innerHTML = config.members.map(m => `<option value="${m}" ${data.paidBy === m ? 'selected':''}>${m}</option>`).join('');


      // Build Split Weight Inputs

      const sharesContainer = document.getElementById('split-shares-container');

      const totalAmount = parseFloat(data.amountAUD) || 0;

      

      // If new, generate equal split weights default matrix

      let currentSplits = data.splits || {};

      if (Object.keys(currentSplits).length === 0) {

        const share = totalAmount / config.members.length;

        config.members.forEach(m => { currentSplits[m] = parseFloat(share.toFixed(2)); });

      }


      sharesContainer.innerHTML = config.members.map(m => `

        <div class="flex items-center justify-between gap-2">

          <span class="text-xs font-medium text-slate-600">${m} Share (A$)</span>

          <input type="number" step="0.01" data-member="${m}" class="split-member-input w-24 p-1 text-right border border-slate-300 rounded text-xs" value="${currentSplits[m] || 0}">

        </div>

      `).join('');


      setupSplitListeners();

      recalculateSplitTotal();

    }

    function syncItinerary() {

      onSnapshot(collection(db, "sydney_itinerary"), (snapshot) => {

        const tableBody = document.getElementById('itinerary-table-body');

        let rows = [];

        snapshot.forEach((doc) => { rows.push({ id: doc.id, ...doc.data() }); });

        

        // Chronological sorting metrics

        rows.sort((a, b) => (a.order || 0) - (b.order || 0));


        if(rows.length === 0) {

          tableBody.innerHTML = `<tr><td colspan="4" class="p-6 text-center text-slate-400">Itinerary Empty. Press "+ Add Row" to begin planning.</td></tr>`;

          return;

        }


        tableBody.innerHTML = rows.map(row => `

          <tr class="hover:bg-slate-50 transition" data-id="${row.id}">

            <td class="p-2 border-b border-slate-100">

              <input type="text" value="${row.dateTime || ''}" onblur="window.updateItineraryField('${row.id}', 'dateTime', this.value)" class="w-full bg-transparent p-1 focus:bg-white focus:ring-1 focus:ring-indigo-500 rounded border-0 outline-0">

            </td>

            <td class="p-2 border-b border-slate-100">

              <input type="text" value="${row.hubLocation || ''}" onblur="window.updateItineraryField('${row.id}', 'hubLocation', this.value)" class="w-full bg-transparent p-1 font-bold text-slate-700 focus:bg-white focus:ring-1 focus:ring-indigo-500 rounded border-0 outline-0">

            </td>

            <td class="p-2 border-b border-slate-100">

              <input type="text" value="${row.activityDetails || ''}" onblur="window.updateItineraryField('${row.id}', 'activityDetails', this.value)" class="w-full bg-transparent p-1 focus:bg-white focus:ring-1 focus:ring-indigo-500 rounded border-0 outline-0">

            </td>

            <td class="p-2 border-b border-slate-100 text-center">

              <button onclick="window.deleteItineraryRow('${row.id}')" class="text-red-400 hover:text-red-600 font-bold text-xs">✕</button>

            </td>

          </tr>

        `).join('');

      });

    }
